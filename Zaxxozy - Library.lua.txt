local function tween(obj, t, props, style, dir)
	local tw = TweenService:Create(obj, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)
	tw:Play()
	return tw
end

local function new(class, props, children)
	local o = Instance.new(class)
	local parent
	for k, v in pairs(props or {}) do
		if k == "Parent" then
			parent = v
		else
			o[k] = v
		end
	end
	for _, c in ipairs(children or {}) do
		c.Parent = o
	end
	if parent then
		o.Parent = parent
	end
	return o
end

local function corner(o, r)
	if typeof(r) == "number" then
		r = UDim.new(0, r)
	end
	return new("UICorner", { CornerRadius = r, Parent = o })
end

local function stroke(o, color, thickness, transparency)
	return new("UIStroke", {
		Color = color,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		Parent = o,
	})
end

local FULL = UDim.new(1, 0)

local function makeMagnifier(parent, color, size)
	local th = math.max(1.5, size / 9)
	local holder = new("Frame", { BackgroundTransparency = 1, Size = UDim2.fromOffset(size, size), Parent = parent })
	local circle = new("Frame", { BackgroundTransparency = 1, Size = UDim2.fromScale(0.62, 0.62), Parent = holder })
	corner(circle, FULL)
	stroke(circle, color, th)
	local handle = new("Frame", {
		BackgroundColor3 = color,
		BorderSizePixel = 0,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.8, 0.8),
		Size = UDim2.fromOffset(size * 0.4, th),
		Rotation = 45,
		Parent = holder,
	})
	corner(handle, FULL)
	return holder
end

local function firstChar(s)
	local ok, ch = pcall(function()
		return utf8.char(utf8.codepoint(s, 1))
	end)
	if ok and ch then
		return string.upper(ch)
	end
	return string.upper(string.sub(s, 1, 1))
end

local function matchScore(q, text)
	if q == "" then
		return 1
	end
	text = string.lower(text)
	q = string.lower(q)
	if text == q then
		return 100
	end
	local s = string.find(text, q, 1, true)
	if s == 1 then
		return 90
	elseif s then
		return 70
	end
	-- fuzzy: ตัวอักษรเรียงตามลำดับ
	local ok, res = pcall(function()
		local qc = {}
		for _, c in utf8.codes(q) do
			table.insert(qc, c)
		end
		local idx = 1
		for _, c in utf8.codes(text) do
			if c == qc[idx] then
				idx += 1
				if idx > #qc then
					return 40
				end
			end
		end
		return 0
	end)
	return ok and res or 0
end

local function getGuiParent()
	local ok, ui = pcall(function()
		return gethui and gethui()
	end)
	if ok and ui then
		return ui
	end
	ok, ui = pcall(function()
		return game:GetService("CoreGui")
	end)
	if ok and ui then
		return ui
	end
	return LocalPlayer:WaitForChild("PlayerGui")
end

local function loadRemoteImage(url, fileName)
	local ok, res = pcall(function()
		local req = request or http_request or (syn and syn.request)
		if not (req and writefile and getcustomasset) then
			error("unsupported")
		end
		local r = req({ Url = url, Method = "GET" })
		if r and r.Body and #r.Body > 0 then
			writefile(fileName, r.Body)
			return getcustomasset(fileName)
		end
		error("failed")
	end)
	if ok then
		return res
	end
	return nil
end

function Window:SetToggleColors(onColor, offColor)
	if onColor then self._toggleOn = onColor end
	if offColor then self._toggleOff = offColor end
end

function Window:_register(item)
	table.insert(self._items, item)
end

function Window:SelectCategory(cat)
	for _, c in ipairs(self.Categories) do
		local sel = (c == cat)
		c.Page.Visible = sel
		tween(c.Button, 0.2, { BackgroundTransparency = sel and 0 or 1 })
		c._stroke.Enabled = sel
		tween(c.Label, 0.2, { TextColor3 = sel and Theme.Text or Theme.SubText })
	end
	self.Current = cat
end

function Window:GoTo(target)
	self:CloseSearch()
	local cat, item
	if target.Page then
		cat = target
	else
		item = target
		cat = item.Category
	end
	self:SelectCategory(cat)
	task.spawn(function()
		RunService.Heartbeat:Wait()
		RunService.Heartbeat:Wait()
		local page = cat.Page
		if item then
			local y = item.Row.AbsolutePosition.Y - page.AbsolutePosition.Y + page.CanvasPosition.Y - 8
			local maxScroll = math.max(0, page.AbsoluteCanvasSize.Y - page.AbsoluteWindowSize.Y)
			y = math.clamp(y, 0, maxScroll)
			tween(page, 0.45, { CanvasPosition = Vector2.new(0, y) })
			task.wait(0.5)
			-- ไฮไลต์สีขาววาบแล้วหายไป
			local f = item.Flash
			for _ = 1, 2 do
				tween(f, 0.18, { BackgroundTransparency = 0.55 })
				task.wait(0.2)
				tween(f, 0.35, { BackgroundTransparency = 1 })
				task.wait(0.38)
			end
		else
			tween(page, 0.35, { CanvasPosition = Vector2.new(0, 0) })
		end
	end)
end

function Category:AddToggle(a, b, c, d, e)
	local o = type(a) == "table" and a or { Name = a, Default = b, Callback = c, OnColor = d, OffColor = e }
	local win = self.Window
	local toggle = {
		Name = o.Name or "Toggle",
		Value = false,
		OnColor = o.OnColor or win._toggleOn,
		OffColor = o.OffColor or win._toggleOff,
		Callback = o.Callback or function() end,
		Category = self,
		Kind = "สวิตช์",
	}
	local row, flash = makeRow(self, 40)
	toggle.Row, toggle.Flash = row, flash
	new("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 14, 0, 0),
		Size = UDim2.new(1, -80, 1, 0),
		Text = toggle.Name,
		TextColor3 = Theme.Text,
		Font = Enum.Font.GothamMedium,
		TextSize = 14,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = row,
	})
	local track = new("Frame", {
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.fromOffset(46, 26),
		BackgroundColor3 = toggle.OffColor,
		Parent = row,
	})
	corner(track, FULL)
	stroke(track, Color3.new(0, 0, 0), 1, 0.7)
	local knob = new("Frame", {
		Position = UDim2.new(0, 3, 0.5, -10),
		Size = UDim2.fromOffset(20, 20),
		BackgroundColor3 = Color3.fromRGB(246, 246, 250),
		Parent = track,
	})
	corner(knob, FULL)
	stroke(knob, Color3.new(0, 0, 0), 1, 0.85)
	local hit = new("TextButton", {
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		Text = "",
		ZIndex = 5,
		Parent = row,
	})

	local function render(animate)
		local t = animate and 0.22 or 0
		tween(track, t, { BackgroundColor3 = toggle.Value and toggle.OnColor or toggle.OffColor })
		tween(knob, t, { Position = toggle.Value and UDim2.new(1, -23, 0.5, -10) or UDim2.new(0, 3, 0.5, -10) }, Enum.EasingStyle.Quint)
	end
	function toggle:Set(v, silent)
		self.Value = v and true or false
		render(true)
		if not silent then
			task.spawn(function() pcall(self.Callback, self.Value) end)
		end
	end
	function toggle:Get() return self.Value end
	function toggle:SetColors(on, off)
		if on then self.OnColor = on end
		if off then self.OffColor = off end
		render(true)
	end
	hit.MouseButton1Click:Connect(function() toggle:Set(not toggle.Value) end)
	hit.MouseEnter:Connect(function() tween(row, 0.15, { BackgroundColor3 = Color3.fromRGB(40, 40, 50) }) end)
	hit.MouseLeave:Connect(function() tween(row, 0.15, { BackgroundColor3 = Theme.Item }) end)

	toggle.Value = o.Default and true or false
	render(false)
	if toggle.Value then
		task.spawn(function() pcall(toggle.Callback, toggle.Value) end)
	end

	table.insert(self.Items, toggle)
	win:_register(toggle)
	return toggle
end

function Category:AddButton(a, b)
	local o = type(a) == "table" and a or { Name = a, Callback = b }
	local btn = { Name = o.Name or "Button", Callback = o.Callback or function() end, Category = self, Kind = "ปุ่ม" }
	local row, flash = makeRow(self, 40)
	btn.Row, btn.Flash = row, flash
	new("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 14, 0, 0),
		Size = UDim2.new(1, -50, 1, 0),
		Text = btn.Name,
		TextColor3 = Theme.Text,
		Font = Enum.Font.GothamMedium,
		TextSize = 14,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = row,
	})
	new("TextLabel", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.fromOffset(16, 24),
		Text = "›",
		TextColor3 = Theme.SubText,
		Font = Enum.Font.GothamBold,
		TextSize = 22,
		Parent = row,
	})
	local hit = new("TextButton", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "", ZIndex = 5, Parent = row })
	hit.MouseButton1Down:Connect(function() tween(row, 0.1, { BackgroundColor3 = Color3.fromRGB(50, 50, 64) }) end)
	hit.MouseButton1Up:Connect(function() tween(row, 0.2, { BackgroundColor3 = Theme.Item }) end)
	hit.MouseLeave:Connect(function() tween(row, 0.2, { BackgroundColor3 = Theme.Item }) end)
	hit.MouseButton1Click:Connect(function() task.spawn(function() pcall(btn.Callback) end) end)
	table.insert(self.Items, btn)
	self.Window:_register(btn)
	return btn
end

function Category:AddLabel(text)
	local lbl = new("TextLabel", {
		Size = UDim2.new(1, 0, 0, 26),
		BackgroundTransparency = 1,
		Text = tostring(text),
		TextColor3 = Theme.SubText,
		Font = Enum.Font.GothamMedium,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left,
		LayoutOrder = #self.Items + 1,
		Parent = self.Page,
	})
	new("UIPadding", { PaddingLeft = UDim.new(0, 6), Parent = lbl })
	local obj = { Frame = lbl }
	function obj:Set(t) lbl.Text = tostring(t) end
	table.insert(self.Items, obj)
	return obj
end

Category.Label = Category.AddLabel
